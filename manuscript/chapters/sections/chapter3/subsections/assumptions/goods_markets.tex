\subsection{財市場}
\label{sec:chapter3_assumptions_goods_markets}
自国および外国の財市場は均衡するものと仮定する。

\paragraph{自国の財市場}
\label{sec:chapter3_assumptions_goods_markets_home}
自国家計\(h\)の財の生産\(y_t^h\)と
世界の消費\(\sum_{h'\in H}c_t^{h'\to h}+\sum_{f\in F}c_t^{f\to h}\)が一致する。
\begin{equation}
\label{form:chapter3_assumptions_goods_markets_home_eq}
    y_t^h=\sum_{h'\in H}c_t^{h'\to h}+\sum_{f\in F}c_t^{f\to h}
\end{equation}

\paragraph{外国の財市場}
\label{sec:chapter3_assumptions_goods_markets_foreign}
同様に
外国家計\(f\)の財の生産\(y_t^f\)と
世界の消費\(\sum_{h\in H}c_t^{h\to f}+\sum_{f'\in F}c_t^{f'\to f}\)が一致する。
\begin{equation}
\label{form:chapter3_assumptions_goods_markets_foreign_eq}
    y_t^f=\sum_{h\in H}c_t^{h\to f}+\sum_{f'\in F}c_t^{f'\to f}
\end{equation}
