\subsection{金融政策}
\label{sec:chapter3_policies_monetary}
本稿において変数\(x\)のハット\(\hat{x}\)は定常状態からの対数乖離を表し、以下で定義される。
\begin{equation}
\label{sec:chapter3_policies_monetary_hat_def}
    \hat{x}_t^H=\log(x_t^H)-\log(x_{ss}^H)
\end{equation}
この対数乖離は乖離が微小である場合に限り
その変数の変化率\((y_t^H-y_{ss}^H)/y_{ss}^H\)の良い近似となるため広く用いられる。
グロス・インフレ率\(\pi_t^H\)およびネット・インフレ率\(\bar{\pi}_t^H\)は以下のように定義される。
\begin{equation}
\label{form:chapter3_policies_monetary_pi_H_def}
    \pi_t^H\equiv\frac{p_t^H}{p_{t-1}^H}
\end{equation}
\begin{equation}
\label{form:chapter3_policies_monetary_pi_H_bar_def}
    \bar{\pi}_t^H\equiv\pi_t^H-1
\end{equation}
定常状態におけるグロス・インフレ率\(\pi_{ss}^H\)は\(1\)である。
したがってグロス・インフレ率の対数乖離\(\hat{\pi}_t^H\)は以下のように計算される。
\begin{equation}
\label{form:chapter3_policies_monetary_pi_H_eq}
    \begin{aligned}
        \hat{\pi}_t^H&\equiv\log(\pi_t^H)-\log(\pi_{ss}^H) \\
        &=\log(\pi_t^H)-\log(1) \\
        &=\log\left(\frac{p_t^H}{p_{t-1}^H}\right) \\
        &=\log(p_t^H)-\log(p_{t-1}^H)
    \end{aligned}
\end{equation}

\subsubsection{自国の金融政策}
\label{sec:chapter3_policies_monetary_home}
自国の中央銀行は利子率\(i_t^H\)を操作する。
本稿ではシミュレーションにおいて自国が従う以下の11種類の政策ルールを比較分析する。
すべての変数は一人当たりの変数で記述される。

\paragraph{1.総消費水準目標(CLT)}
\label{sec:chapter3_policies_monetary_home_clt}
利子率は実質総消費（\(c_t^{H\to W}\)）のあらかじめ定められた目標パス\(\chi_t^H\)からの対数乖離（\(\log(c_t^{H\to W})-\log(\chi_t^H)\)）に反応し、さらに過去の乖離の累積\(\gamma_t^H\)にも反応する。ここで\(i_{ss}^H\)は利子率の定常状態の値である。
\begin{align}
\label{form:chapter3_policies_monetary_home_clt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(c_t^{H\to W})-\log(\chi_t^H))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_home_clt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(c_{t-1}^{H\to W})-\log(\chi_{t-1}^H))
\end{align}
ここで\(\phi_{gap}^H, \phi_{level}^H\)および\(\chi_t^H\)は
各水準目標（LT）において、値は異なるものの、共通して用いられる性質のものである。
まず\(\phi_{gap}^H, \phi_{level}^H\)は任意値のパラメータであり、本稿ではこれらを、
\(\beta\)ショックが起こった際に厚生\(W^H\)の落ち込みを最小にするようなものに決める
（\ref{sec:chapter4parameters_monetary_optimization}節）。
\(\chi_t^H\)は目標パスであり、
定常状態\(\chi_{ss}^H\)を基準とした1次自己回帰（AR(1)）過程に従って以下のように変動すると仮定する。
\begin{equation}
\label{form:chapter3_policies_monetary_home_clt_chi_H_eq}
    \log(\chi_t^H)-\log(\chi_{ss}^H)=\rho_{\chi}^H\left(\log(\chi_{t-1}^H)-\log(\chi_{ss}^H)\right)+\varepsilon_t^{\chi,H}
\end{equation}

\paragraph{2.消費者物価水準目標(CPLT)}
\label{sec:chapter3_policies_monetary_home_cplt}
利子率は消費者物価水準（\(p_t^{H\to W}\)）の目標パス\(\chi_t^H\)からの対数乖離とその累積に反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_cplt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(p_t^{H\to W})-\log(\chi_t^H))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_cplt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(p_{t-1}^{H\to W})-\log(\chi_{t-1}^H))
\end{align}

\paragraph{3.消費者物価インフレ目標(IT-CPI)}
\label{sec:chapter3_policies_monetary_home_it_cpi}
利子率は消費者物価指数を用いたグロス・インフレ率（\(\pi_t^{H\to W}\)）の定常状態からの乖離に反応する。
\begin{equation}
\label{form:chapter3_policies_monetary_home_it_cpi_i_H_eq}
    i_t^H=i_{ss}^H+\phi_{\pi}^H(\pi_t^{H\to W}-\pi_{ss}^{H\to W})+\varepsilon_t^{i,H}
\end{equation}

\paragraph{4.生産者物価インフレ目標(IT-PPI)}
\label{sec:chapter3_policies_monetary_home_it_ppi}
利子率\(i_t^H\)は生産者物価指数を用いたグロス・インフレ率（\(\pi_t^H\)）の定常状態からの乖離に反応する。
\begin{equation}
\label{form:chapter3_policies_monetary_home_it_ppi_i_H_eq}
    i_t^H=i_{ss}^H+\phi_{\pi}^H(\pi_t^H-\pi_{ss}^H)+\varepsilon_t^{i,H}
\end{equation}

\paragraph{5.名目総消費水準目標(NCLT)}
\label{sec:chapter3_policies_monetary_home_nclt}
利子率は一人当たりの名目総消費（\(p_t^{H\to W}c_t^{H\to W}\)）の目標パス\(\chi_t^H\)からの対数乖離と
その累積に反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_nclt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(p_t^{H\to W}c_t^{H\to W})-\log(\chi_t^H))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_home_nclt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(p_{t-1}^{H\to W}c_{t-1}^{H\to W})-\log(\chi_{t-1}^H))
\end{align}

\paragraph{6.名目GDP水準目標(NGDPLT)}
\label{sec:chapter3_policies_monetary_home_ngdplt}
利子率は一人当たりの名目GDP（\(p_t^Hy_t^H\)）の目標パス\(\chi_t^H\)からの対数乖離とその累積に反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_ngdplt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(p_t^Hy_t^H)-\log(\chi_t^H))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_home_ngdplt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(p_{t-1}^Hy_{t-1}^H)-\log(\chi_{t-1}^H))
\end{align}

\paragraph{7.生産水準目標(OLT)}
\label{sec:chapter3_policies_monetary_home_olt}
利子率は実質産出量（\(y_t^H\)）の目標パス\(\chi_t^H\)からの対数乖離とその累積に反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_olt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(y_t^H)-\log(\chi_t^H))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_home_olt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(y_{t-1}^H)-\log(\chi_{t-1}^H))
\end{align}

\paragraph{8.潜在生産水準目標(POLT)}
\label{sec:chapter3_policies_monetary_home_polt}
利子率は実質産出量（\(y_t^H\)）と潜在産出量（\(y_t^{H,pot}\)）の対数乖離（産出量ギャップ）、および過去のギャップの累積に反応する。本ルールでは外生的な目標パスの代わりに潜在産出量との乖離が指標として用いられる。
\begin{align}
\label{form:chapter3_policies_monetary_home_polt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(y_t^H)-\log(y_t^{H,pot}))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_home_polt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(y_{t-1}^H)-\log(y_{t-1}^{H,pot}))
\end{align}

\paragraph{9.生産者物価水準目標(PPLT)}
\label{sec:chapter3_policies_monetary_home_pplt}
利子率は生産者物価水準（\(p_t^H\)）のあらかじめ定められた目標パス\(\chi_t^H\)からの対数乖離とその累積\(\gamma_t^H\)に反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_pplt_i_H_eq}
    i_t^H&=i_{ss}^H+\phi_{gap}^H(\log(p_t^H)-\log(\chi_t^H))+\phi_{level}^H\gamma_t^H \\
\label{form:chapter3_policies_monetary_home_pplt_gamma_H_eq}
    \gamma_t^H&=\gamma_{t-1}^H+(\log(p_{t-1}^H)-\log(\chi_{t-1}^H))
\end{align}

\paragraph{10.消費者物価テイラー・ルール(TR-CPI)}
\label{sec:chapter3_policies_monetary_home_tr_cpi}
利子率は金利平滑化を伴い、消費者物価インフレ率（\(\pi_t^{H\to W}\)）の対数、および産出量ギャップに反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_tr_cpi_i_H_notional_eq}
    i_t^{H,notional}&=\rho_i^Hi_{t-1}^{H,notional}+(1-\rho_i^H)\left(i_{ss}^H+\phi_{\pi}^H\log(\pi_t^{H\to W})+\phi_y^H(\log(y_t^H)-\log(y_t^{H,pot}))\right)+\varepsilon_t^{i,H} \\
\label{form:chapter3_policies_monetary_home_tr_cpi_i_H_eq}
    i_t^H&=i_t^{H,notional}
\end{align}

\paragraph{11.生産者物価テイラー・ルール(TR-PPI)}
\label{sec:chapter3_policies_monetary_tr_ppi}
利子率は金利平滑化（スムージング）を伴い、生産者物価インフレ率（\(\pi_t^H\)）の対数、および実質産出量（\(y_t^H\)）と潜在産出量（\(y_t^{H,pot}\)）の対数乖離に反応する。
\begin{align}
\label{form:chapter3_policies_monetary_home_tr_ppi_i_H_notional_eq}
    i_t^{H,notional}&=\rho_i^Hi_{t-1}^{H,notional}+(1-\rho_i^H)\left(i_{ss}^H+\phi_{\pi}^H\log(\pi_t^H)+\phi_y^H(\log(y_t^H)-\log(y_t^{H,pot}))\right)+\varepsilon_t^{i,H} \\
\label{form:chapter3_policies_monetary_home_tr_ppi_i_H_eq}
    i_t^H&=i_t^{H,notional}
\end{align}

\subsubsection{外国の金融政策}
\label{sec:chapter3_policies_monetary_foreign}
外国の中央銀行は本稿の分析を通じて一貫して以下の政策ルールに従うものと仮定する。
シミュレーションにおいて比較対象とするのは自国の政策ルールのみである。

\paragraph{生産者物価インフレ目標(IT-PPI)}
\label{sec:chapter3_policies_monetary_foreign_it_ppi}
外国の利子率\(i_t^F\)は、外国の生産者物価指数を用いたグロス・インフレ率（\(\pi_t^{F*}\)）の定常状態からの乖離に反応する。
\begin{equation}
\label{form:chapter3_policies_monetary_foreign_it_ppi_i_H_eq}
    i_t^F=i_{ss}^F+\phi_{\pi}^F(\pi_t^{F*}-\pi_{ss}^{F*})+\varepsilon_t^{i,F}
\end{equation}
ここで\(i_{ss}^F\)と\(\pi_{ss}^{F*}\)はそれぞれの定常状態の値である。
