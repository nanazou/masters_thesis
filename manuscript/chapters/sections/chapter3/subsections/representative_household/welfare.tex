\subsection{厚生}
\label{sec:chapter3_representative_household_welfare}

\subsubsection{自国の厚生\(W_s^H\)}
\label{sec:chapter3_representative_household_welfare_home}

\paragraph{①自国の元となる方程式}
\label{sec:chapter3_representative_household_welfare_home_original}
自国家計\(h\)の厚生\(W_s^h\)は、期待生涯効用関数の定常状態からの乖離\(U_t^h-u_{ss}^h\)となるよう以下により定義される。
\begin{equation}
\label{form:chapter3_representative_household_welfare_home_original_W_h_def}
    W_s^h\equiv\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^H\right)\left[u_t^h-u_{ss}^h\right]
\end{equation}
ここで\(t=s\)期における割引項は\(\prod_{k=s}^{s-1}\beta_k^H=1\)である。

\paragraph{②自国代表的家計の方程式}
\label{sec:chapter3_representative_household_welfare_home_representative}
式\eqref{form:chapter3_representative_household_utility_home_representative_average_u_h_modified_eq}および\eqref{form:chapter3_representative_household_utility_home_representative_u_H_def}より、自国家計の効用の平均\(\frac{1}{N}\sum_{h\in H}u_t^h\)は以下のようになるのであった。
\begin{equation}
\label{form:chapter3_representative_household_welfare_home_representative_u_H_eq}
    \frac{1}{N}\sum_{h\in H}u_t^h\approx u_t^H
\end{equation}
このことに注意しながら式\eqref{form:chapter3_representative_household_welfare_home_original_W_h_def}を\(h\)について足し合わせ、両辺を自国の人口\(N\)で割ることで、自国家計の厚生の平均\(\frac{1}{N}\sum_{h\in H}W_s^h\)を求める。
\begin{equation}
\label{form:chapter3_representative_household_welfare_home_representative_average_W_h_eq}
\begin{aligned}
    \frac{1}{N}\sum_{h\in H}W_s^h
    &=\frac{1}{N}\sum_{h\in H}\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^H\right)\left[u_t^h-u_{ss}^h\right] \\
    &=\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^H\right)\left[\frac{1}{N}\sum_{h\in H}u_t^h-u_{ss}^h\right] \\
    &\approx\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^H\right)\left[u_t^H-u_{ss}^H\right]
\end{aligned}
\end{equation}
そこでこの右辺によって代表的家計の厚生\(W_s^H\)を定義する。
\begin{equation}
\label{form:chapter3_representative_household_welfare_home_representative_W_H_def}
    W_s^H\equiv\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^H\right)\left[u_t^H-u_{ss}^H\right]
\end{equation}

\subsubsection{外国の厚生\(W_s^F\)}

\paragraph{②外国代表的家計の方程式}
\label{sec:chapter3_representative_household_welfare_foreign_representative}
式\eqref{form:chapter3_representative_household_utility_foreign_representative_average_u_f_modified_eq}および\eqref{form:chapter3_representative_household_utility_foreign_representative_u_F_def}より、外国家計の効用の平均\(\frac{1}{M}\sum_{f\in F}u_t^f\)は以下のようになるのであった。
\begin{equation}
\label{form:chapter3_representative_household_welfare_foreign_representative_u_F_eq}

\end{equation}
このことに注意しながら式\eqref{form:chapter3_representative_household_welfare_foreign_original_W_f_def}を\(f\)について足し合わせ、両辺を外国の人口\(M\)で割ることで、外国家計の厚生の平均\(\frac{1}{M}\sum_{f\in F}W_s^f\)を求める。
\begin{equation}
\label{form:chapter3_representative_household_welfare_foreign_representative_average_W_f_eq}
\begin{aligned}
    \frac{1}{M}\sum_{f\in F}W_s^f
    &=\frac{1}{M}\sum_{f\in F}\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^F\right)\left[u_t^f-u_{ss}^f\right] \\
    &=\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^F\right)\left[\frac{1}{M}\sum_{f\in F}u_t^f-u_{ss}^f\right] \\
    &\approx\operatorname{E}_s\sum_{t=s}^{\infty}\left(\prod_{k=s}^{t-1}\beta_k^F\right)\left[u_t^F-u_{ss}^F\right]
\end{aligned}
\end{equation}

\label{form:chapter3_representative_household_welfare_foreign_representative_W_F_def}

\end{equation}
