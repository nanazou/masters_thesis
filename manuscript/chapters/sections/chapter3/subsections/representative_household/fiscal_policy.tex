\subsection{財政政策}
\label{sec:chapter3_representative_household_fiscal_policy}

\subsubsection{自国の財政政策}
\label{sec:chapter3_representative_household_fiscal_policy_home}

\paragraph{①自国の元となる方程式}
\label{sec:chapter3_representative_household_fiscal_policy_home_original}
式
\eqref{form:chapter3_policies_fiscal_home_budget_modified_eq}
で示されたとおり、政府の予算制約は以下のようであった。
\begin{equation}
\label{form:chapter3_representative_household_fiscal_policy_home_original_budget_eq}
    N\mathrm{t}_t^H=\tau_t^Hp_t^HY_t^H
\end{equation}

\paragraph{②自国代表的家計の方程式}
\label{sec:chapter3_representative_household_fiscal_policy_home_representative}
定義
\eqref{form:chapter3_representative_household_resource_constraints_home_y_H_def}
\(y_t^H\equiv Y_t^H/N\)を上式に代入する。
\begin{equation}
\label{form:chapter3_representative_household_fiscal_policy_home_representative_budget_eq}
    N\mathrm{t}_t^H=\tau_t^Hp_t^H(Ny_t^H)
\end{equation}
この式の両辺を人口\(N\)で割ることで一括還付\(\mathrm{t}_t^H\)が得られる。
\begin{equation}
\label{form:chapter3_representative_household_fiscal_policy_home_representative_budget2_eq}
    \mathrm{t}_t^H=\tau_t^Hp_t^Hy_t^H
\end{equation}

\subsubsection{外国の財政政策}
\label{sec:chapter3_representative_household_fiscal_policy_foreign}

\paragraph{①外国の元となる方程式}
\label{sec:chapter3_representative_household_fiscal_policy_foreign_original}
式
\eqref{form:chapter3_policies_fiscal_foreign_budget_modified_eq}
で示されたとおり、外国政府の予算制約は以下のようであった。
\begin{equation}
\label{form:chapter3_representative_household_fiscal_policy_foreign_original_budget_eq}
    M\mathrm{t}_t^{F*}=\tau_t^Fp_t^{F*}Y_t^F
\end{equation}

\paragraph{②外国の代表的家計の方程式}
\label{sec:chapter3_representative_household_fiscal_policy_foreign_representative}
定義
\eqref{form:chapter3_representative_household_resource_constraints_foreign_y_F_def}
\(y_t^F\equiv Y_t^F/M\)を上式に代入する。
\begin{equation}
\label{form:chapter3_representative_household_fiscal_policy_foreign_representative_budget_eq}
    M\mathrm{t}_t^{F*}=\tau_t^Fp_t^{F*}(My_t^F)
\end{equation}
この式の両辺を人口\(M\)で割ることで一括還付\(\mathrm{t}_t^{F*}\)が得られる。
\begin{equation}
\label{form:chapter3_representative_household_fiscal_policy_foreign_representative_budget2_eq}
    \mathrm{t}_t^{F*}=\tau_t^Fp_t^{F*}y_t^F
\end{equation}
