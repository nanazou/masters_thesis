\subsection{所得の限界効用}
\label{sec:chapter3_representative_household_lambda}

\subsubsection{自国の所得の限界効用}
\label{sec:chapter3_representative_household_lambda_home}

\paragraph{①自国の元となる方程式}
\label{sec:chapter3_representative_household_lambda_home_original}
式
\eqref{form:chapter3_households_stage2_home_lambda_h_p_H_W_c_h_W_eq}
で示されたとおり、自国家計\(h\)の総消費指数\(c_t^{h\to W}\)に関するFOCは以下のようであった。
\begin{equation}
\label{form:chapter3_representative_household_lambda_home_original_lambda_h_p_H_W_c_h_W_eq}
    \lambda_t^h=\frac{1}{p_t^{H\to W}c_t^{h\to W}}
\end{equation}

\paragraph{②自国代表的家計の方程式}
\label{sec:chapter3_representative_household_lambda_home_representative}
式
\eqref{form:chapter3_risk_sharing_commonization_home_lambda_h_equality_eq}および
\eqref{form:chapter3_risk_sharing_commonization_home_c_h_W_equality_eq}
で示されたとおり、すべての自国家計\(h\)について\(\lambda_t^h\)と\(c_t^{h\to W}\)は共通の値をとる。
そこで代表的家計の所得の限界効用\(lambda_t^H\)および総消費指数\(c_t^{H\to W}\)を以下のように定義する。
\begin{equation}
\label{form:chapter3_representative_household_lambda_home_representative_lambda_H_def}
    \lambda_t^H\equiv\lambda_t^h
\end{equation}
\begin{equation}
\label{form:chapter3_representative_household_lambda_home_representative_c_H_W_def}
    c_t^{H\to W}\equiv c_t^{h\to W}
\end{equation}
これらを式
\eqref{form:chapter3_representative_household_lambda_home_original_lambda_h_p_H_W_c_h_W_eq}
に代入すると自国代表的家計の所得の限界効用の式が得られる。
\begin{equation}
\label{form:chapter3_representative_household_lambda_home_representative_lambda_H_p_H_W_c_H_W_eq}
    \lambda_t^H=\frac{1}{p_t^{H\to W}c_t^{H\to W}}
\end{equation}

\subsubsection{外国の所得の限界効用}
\label{sec:chapter3_representative_household_lambda_foreign}

\paragraph{①外国の元となる方程式}
\label{sec:chapter3_representative_household_lambda_foreign_original}
式
\eqref{form:chapter3_households_stage2_foreign_lambda_f_slash_star_p_F_W_star_c_f_W_eq}
で示されたとおり、外国家計\(f\)の総消費指数\(c_t^{f\to W}\)に関するFOCは以下のようであった。
\begin{equation}
\label{form:chapter3_representative_household_lambda_foreign_original_lambda_f_slash_star_p_F_W_star_c_f_W_eq}
    \lambda_t^{f/*}=\frac{1}{p_t^{F\to W*}c_t^{f\to W}}
\end{equation}

\paragraph{②外国代表的家計の方程式}
\label{sec:chapter3_representative_household_lambda_foreign_representative}
式
\eqref{form:chapter3_risk_sharing_commonization_foreign_lambda_f_slash_star_equality_eq}および
\eqref{form:chapter3_risk_sharing_commonization_foreign_c_f_W_equality_eq}
で示されたとおり、すべての外国家計\(f\)について\(\lambda_t^{f/*}\)と\(c_t^{f\to W}\)は共通の値をとる。
そこで代表的家計の変数を以下のように定義する。
\begin{equation}
\label{form:chapter3_representative_household_lambda_foreign_representative_lambda_F_slash_star_def}
    \lambda_t^{F/*}\equiv\lambda_t^{f/*}
\end{equation}
\begin{equation}
\label{form:chapter3_representative_household_lambda_foreign_representative_c_F_W_def}
    c_t^{F\to W}\equiv c_t^{f\to W}
\end{equation}
これらを式
\eqref{form:chapter3_representative_household_lambda_foreign_original_lambda_f_slash_star_p_F_W_star_c_f_W_eq}
に代入すると外国代表的家計の所得の限界効用の式が得られる。
\begin{equation}
\label{form:chapter3_representative_household_lambda_foreign_original_lambda_F_slash_star_p_F_W_star_c_F_W_eq}
    \lambda_t^{F/*}=\frac{1}{p_t^{F\to W*}c_t^{F\to W}}
\end{equation}
