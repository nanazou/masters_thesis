\subsubsection{各政策の分析の概要}
\label{chapter5_beta_shock_policies_overview}
ここから各政策について、厚生\(W_s^H\)（\(\Delta_t^H\)有り）の落ち込みが小さい（評価が高い）順に
4曲線などを用いた分析をおこなう。
厚生\(W^H\)の式
\eqref{form:chapter5_beta_shock_welfare_W_H_substituted_eq}
によれば、その決定要因は以下の4つの変数であった。
\begin{enumerate}
    \item
    \label{itm:chapter5_beta_shock_policies_overview_c_H_W}
        \(c_t^{H \to W}\)：
        厚生\(W^H\)と正の相関がある。
    \item
    \label{itm:chapter5_beta_shock_policies_overview_y_H}
        \(y_t^H\)：
        厚生\(W^H\)と負の相関がある。
    \item
    \label{itm:chapter5_beta_shock_policies_overview_pi_H}
        \(\pi_t^H\)：
        定常状態からの乖離が厚生\(W^H\)と負の相関がある。
    \item
    \label{itm:chapter5_beta_shock_policies_overview_Delta_t_minus_1_H}
        \(\Delta_{t-1}^H\)：
        厚生\(W^H\)と負の相関がある。
\end{enumerate}
そこで、各政策が厚生\(W^H\)に与える影響を考えるためにこれらの4つの変数に与える影響について順に議論していく。
この議論のために用いる式を以下に再掲する。
YS-\(\pi\)曲線（差分形式）
\eqref{form:chapter5_two_plane_ys_derivation_pi_H_modified_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_ys_pi_eq}
    \hat{\pi}_t^H-\beta_{ss}^H\operatorname{E}_t[\hat{\pi}_{t+1}^H]=\kappa^H(2\hat{y}_t^H-2\hat{a}_t^H-\hat{X}_t-\hat{\tau}_t^H)
\end{equation}
前方展開されたYS-\(\hat{\pi}\)曲線
\eqref{form:chapter5_two_plane_ys_forward_pi_H_lim_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_ys_pi_H_lim_eq}
    \hat{\pi}_t^H=\operatorname{E}_t\sum_{k=0}^{\infty}(\beta_{ss}^H)^k\kappa^H(2\hat{y}_{t+k}^H-2\hat{a}_{t+k}^H-\hat{X}_{t+k}-\hat{\tau}_{t+k}^H)
\end{equation}
YS-\(\hat{p}\)曲線
\eqref{form:chapter5_two_plane_ys_derivation_p_H_modified_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_ys_p_eq}
    \hat{p}_t^H=\hat{p}_{t-1}^H+\beta_{ss}^H\operatorname{E}_t[\hat{\pi}_{t+1}^H]+\kappa^H(2\hat{y}_t^H-2\hat{a}_t^H-\hat{X}_t-\hat{\tau}_t^H)
\end{equation}
YD曲線
\eqref{form:chapter5_two_plane_yd_connection_derivation_p_H_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_yd_eq}
    p_t^H=\frac{1}{y_t^H}\times\frac{\alpha^H+(1-\alpha^F)\operatorname{E}_t[\mathcal{B}_t\Lambda_{\infty}]}{\lambda_t^H}
\end{equation}
CS-\(\pi\)曲線
\eqref{form:chapter5_two_plane_cs_derivation_pi_H_W_modified_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_cs_pi_eq}
    \hat{\pi}_t^{H\to W}=\beta_{ss}^H\operatorname{E}_t[\hat{\pi}_{t+1}^H]+\kappa^H\hat{\Omega}_t+\frac{1-\alpha^H}{\alpha^H}(\hat{Z}_t-\hat{Z}_{t-1})+\left(\frac{2\kappa^H}{\alpha^H}+\frac{1-\alpha^H}{\alpha^H}\right)\hat{c}_t^{H\to W}-\frac{1-\alpha^H}{\alpha^H}\hat{c}_{t-1}^{H\to W}
\end{equation}
CS-\(\hat{p}\)曲線
\eqref{form:chapter5_two_plane_cs_derivation_p_H_W_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_cs_p_eq}
    \hat{p}_t^{H\to W}=\hat{p}_{t-1}^H+\beta_{ss}^H\operatorname{E}_t[\hat{\pi}_{t+1}^H]+\kappa^H(\hat{\Omega}_t-2\hat{a}_t^H)+\frac{1-\alpha^H}{\alpha^H}\hat{Z}_t+\left(\frac{2\kappa^H}{\alpha^H}+\frac{1-\alpha^H}{\alpha^H}\right)\hat{c}_t^{H\to W}
\end{equation}
CD曲線
\eqref{form:chapter5_two_plane_cd_p_H_W_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_cd_eq}
    p_t^{H\to W}=\frac{1}{c_t^{H\to W}}\times\frac{1}{\beta_t^H(1+i_t^H)\operatorname{E}_t[\lambda_{t+1}^H]}
\end{equation}
\end{equation}
自国コンティンジェント債券に関するオイラー方程式
\eqref{form:chapter3_representative_household_euler_contingent_home_representative_lambda_H_i_H_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_euler_contingent_eq}
    \lambda_t^H=\beta_t^H(1+i_t^H)\operatorname{E}_t[\lambda_{t+1}^H]
\end{equation}
自国コンティンジェント債券に関する前方展開されたオイラー方程式
\eqref{form:chapter3_representative_household_euler_contingent_home_representative_E_lambda_H_i_H_lim_eq}
\begin{equation}
\label{form:chapter5_beta_shock_policies_overview_euler_contingent_lim}
    \begin{aligned}
        \operatorname{E}_t[\lambda_{t+1}^H]
        &=(\lim_{k\to\infty}\lambda_{t+k+1}^H)\operatorname{E}_t\left[\prod_{k=1}^{\infty}\beta_{t+k}^H(1+i_{t+k}^H)\right] \\
        &=(\frac{1}{(\lim_{k\to\infty}p_{t+k+1}^{H\to W})(\lim_{k\to\infty}c_{t+k+1}^{H\to W})})\operatorname{E}_t\left[\prod_{k=1}^{\infty}\beta_{t+k}^H(1+i_{t+k}^H)\right]
        &=(\frac{1}{(\lim_{k\to\infty}p_{t+k+1}^{H\to W})c_{ss}^{H\to W}})\operatorname{E}_t\left[\prod_{k=1}^{\infty}\beta_{t+k}^H(1+i_{t+k}^H)\right]
    \end{aligned}
\end{equation}
